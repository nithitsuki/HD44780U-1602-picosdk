cmake_minimum_required(VERSION 3.13)

# Pull in the Pico SDK (requires PICO_SDK_PATH to be set)
include(pico_sdk_import.cmake)

# ------------------------------------------------------------------
# Project: HD44780U-1602-picosdk
# Native Raspberry Pi Pico driver for the Hitachi HD44780U LCD
# controller (drives a 16x2 character display in 4-bit mode).
# No Arduino libraries — pure pico-sdk + hardware_gpio.
# ------------------------------------------------------------------
project(HD44780U-1602-picosdk
    VERSION 0.1.0
    DESCRIPTION "pico-sdk driver for HD44780U 16x2 LCD"
    LANGUAGES C CXX ASM
)

# Initialise the SDK
pico_sdk_init()

# ==================================================================
# HD44780 static library
# ==================================================================
add_library(hd44780 STATIC
    src/hd44780.c
)

target_include_directories(hd44780 PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/src
)

target_link_libraries(hd44780 PUBLIC
    pico_stdlib
    hardware_gpio
)

# Skip the picotool download — we flash via OpenOCD with .elf files.
set(PICO_NO_PICOTOOL 1)

# ==================================================================
# Examples
# ==================================================================
function(my_pico_add_outputs target)
    pico_add_bin_output(${target})
    pico_add_hex_output(${target})
    pico_add_dis_output(${target})
    # .map is already generated by the linker, no extra step needed.
endfunction()

add_executable(hello_lcd
    examples/hello_lcd/hello_lcd.c
)
target_link_libraries(hello_lcd hd44780)
my_pico_add_outputs(hello_lcd)

add_executable(blink_test
    examples/blink_test/blink_test.c
)
target_link_libraries(blink_test hd44780)
my_pico_add_outputs(blink_test)

add_executable(debug
    examples/debug/debug.c
)
target_link_libraries(debug hd44780)
my_pico_add_outputs(debug)
